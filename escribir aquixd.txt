  void _onItemTapped(int index) {
    setState(() {
      _selectedIndex = index;
    });
  }

  // Función para agregar un nuevo mensaje
  void _enviarMensaje() {
    if (_mensajeController.text.trim().isNotEmpty) {
      setState(() {
        _mensajes.add(_mensajeController.text.trim());
        _mensajeController.clear();
      });
    }
  }

        const Row(

              'no se',

                fontSize: 18,

            ),
          ],
        ),

  // SECCIÓN 2: EVENTOS

      children: [

import 'dart:async';

import 'package:flutter/material.dart';

import 'Preguntas.dart';

class Juego extends StatefulWidget {
  const Juego({super.key});

  @override
  State<Juego> createState() => _JuegoState();
}

class _JuegoState extends State<Juego> {
  Tematica? tematicaActual;

  int indicePregunta = 0;
  int puntos = 0;
  int segundos = 10;

  int? opcionSeleccionada;

  bool respondida = false;
  bool terminado = false;
  bool tiempoAgotado = false;

  Timer? cronometro;
  Timer? avanceAutomatico;

  // INICIAR EL JUEGO

  void iniciarJuego() {
    cronometro?.cancel();
    avanceAutomatico?.cancel();

    setState(() {
      tematicaActual = sortearTematica();

      indicePregunta = 0;
      puntos = 0;
      segundos = 10;

      opcionSeleccionada = null;

      respondida = false;
      terminado = false;
      tiempoAgotado = false;
    });

    iniciarCronometro();
  }

  // CRONÓMETRO

  void iniciarCronometro() {
    cronometro?.cancel();

    cronometro = Timer.periodic(const Duration(seconds: 1), (timer) {
      if (!mounted || respondida || terminado) {
        timer.cancel();
        return;
      }

      if (segundos <= 1) {
        timer.cancel();
        responder(null);
      } else {
        setState(() {
          segundos--;
        });
      }
    });
  }

  // RESPONDER

  void responder(int? opcion) {
    if (respondida || terminado || tematicaActual == null) {
      return;
    }

    cronometro?.cancel();

    final pregunta = tematicaActual!.preguntas[indicePregunta];

    final bool correcta =
        opcion != null && opcion == pregunta.respuestaCorrecta;

    setState(() {
      respondida = true;
      opcionSeleccionada = opcion;
      tiempoAgotado = opcion == null;

      if (tiempoAgotado) {
        segundos = 0;
      }

      if (correcta) {
        puntos += 100;
      }
    });

    // Esperar un momento antes de pasar a la siguiente pregunta.
    avanceAutomatico?.cancel();

    avanceAutomatico = Timer(const Duration(milliseconds: 1300), () {
      if (!mounted) return;

      siguientePregunta();
    });
  }

  // SIGUIENTE PREGUNTA

  void siguientePregunta() {
    if (tematicaActual == null) return;

    if (indicePregunta < tematicaActual!.preguntas.length - 1) {
      setState(() {
        indicePregunta++;

        opcionSeleccionada = null;
        respondida = false;
        tiempoAgotado = false;
        segundos = 10;
      });

      iniciarCronometro();
    } else {
      setState(() {
        terminado = true;
      });
    }
  }

  // NOMBRE DEL PREMIO

  String obtenerPremio() {
    if (puntos >= 1000) {
      return '¡Premio mayor! - Leyenda suprema 🏆';
    } else if (puntos >= 800) {
      return 'Nivel 3 - Aura Pro 🥇';
    } else if (puntos >= 500) {
      return 'Nivel 2 - Aura en desarrollo 🥈';
    } else if (puntos >= 100) {
      return 'Nivel 1 - Noob en entrenamiento 🥉';
    } else {
      return 'Sin premio por ahora ⭐';
    }
  }

  // CANTIDAD DE AURA

  String obtenerAura() {
    if (puntos >= 1000) {
      return '+50.000 de aura';
    } else if (puntos >= 800) {
      return '+10.000 de aura';
    } else if (puntos >= 500) {
      return '+5.000 de aura';
    } else if (puntos >= 100) {
      return '+1.000 de aura';
    } else {
      return '-1.000 de aura';
    }
  }

  // MENSAJE DEL PREMIO

  String obtenerMensajePremio() {
    if (puntos >= 1000) {
      return '¡ERES UNA LEYENDA! Tu aura está en otro nivel. '
          'Nadie puede detenerte.';
    } else if (puntos >= 800) {
      return '¡Qué nivel! Eres casi una leyenda. '
          'El conocimiento fluye por tus venas.';
    } else if (puntos >= 500) {
      return 'Vas por buen camino. Tu potencial '
          'empieza a despertar. Sigue así. ';
    } else if (puntos >= 100) {
      return 'Eres un noob, pero tranqui, '
          'todavía puedes aprender más. ';
    } else {
      return 'Te falta entrenar, pequeño noob.  '
          '¡Vuelve a intentarlo y consigue más aura!';
    }
  }

  // RECLAMAR EL PREMIO

  void reclamarPremio() {
    showDialog<void>(
      context: context,
      builder: (dialogContext) {
        return AlertDialog(
          backgroundColor: const Color(0xFF162B53),
          scrollable: true,
          title: Text(
            puntos >= 100 ? '¡PREMIO DESBLOQUEADO! ' : 'MODO NOOB ACTIVADO ',
            textAlign: TextAlign.center,
            style: const TextStyle(
              color: Color(0xFFEACB74),
              fontWeight: FontWeight.bold,
            ),
          ),
          content: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              const Icon(
                Icons.auto_awesome,

.exit
🗄️ 2. Comprobar MySQL

Primero:

mysql --version

Debería aparecer algo parecido a:

mysql  Ver 8.0.xx

Si Windows dice:

'mysql' no se reconoce...

no necesariamente significa que MySQL esté mal instalado; puede ser que MySQL no esté agregado al PATH.

🔌 3. Comprobar que el servidor MySQL esté encendido

En PowerShell:

Get-Service MySQL*

Podrías ver algo como:

Status   Name
------   ----
Running  MySQL80

Si aparece Stopped, intenta:

Start-Service MySQL80

Y vuelve a comprobar:

Get-Service MySQL*

Debe quedar:

Running

El nombre puede variar según cómo se haya instalado, así que si no existe MySQL80, no ejecutes comandos a ciegas; primero miramos qué nombre te aparece.

🔑 4. Probar conexión a MySQL

Ahora:

mysql -u root -p

Te pedirá:

Enter password:

Escribes la contraseña que pusiste al instalar MySQL.

⚠️ Cuando escribas la contraseña no se verá nada en pantalla. Es normal.

Si entra correctamente, aparecerá algo parecido a:

mysql>

Ahora prueba:

SHOW DATABASES;

Y debería aparecer una lista.

Después:

SELECT VERSION();

Y:

SELECT 1;

Si devuelve:

1

🎉 MySQL está funcionando.

Para salir:

exit;
📁 5. Crear nuestra carpeta del proyecto

mkdir universidad

pwd

code .

🔗 10. Probar Node + MySQL

Ahora reemplaza el contenido de server.js por:

const mysql = require("mysql2");

const conexion = mysql.createConnection({
    host: "localhost",
    user: "root",
    password: "TU_CONTRASEÑA",
    database: "universidad"
});

conexion.connect((error) => {
    if (error) {
        console.error("❌ Error conectando a MySQL:");
        console.error(error.message);
        return;
    }

    console.log("✅ Node.js conectado correctamente a MySQL");

    conexion.query("SELECT 1 AS prueba", (error, resultado) => {
        if (error) {
            console.error("❌ Error ejecutando consulta:");
            console.error(error.message);
            return;
        }

        console.log("Resultado:", resultado);
    });
});

Cambia:

password: "TU_CONTRASEÑA",

por tu contraseña real.

Luego:

node server.js

Si todo está bien deberías obtener algo como:

✅ Node.js conectado correctamente a MySQL
Resultado: [ { prueba: 1 } ]
